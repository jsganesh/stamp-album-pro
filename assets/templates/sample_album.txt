################################################################################
# StampAlbum Pro - Sample Album
#
# A short album in the row-based page description (DSL). Each STAMP_ADD gives
# the stamp's width and height in millimetres, its description, then up to
# three catalogue references. Comments start with # and are ignored.
#
# Catalogue numbers are StampWorld's (Great Britain 1 to 3), the same stamps
# as the app's first-run sample page.
################################################################################

# Document metadata
ALBUM_TITLE ("Great Britain: The Line-Engraved Issues")
ALBUM_AUTHOR ("StampAlbum Pro Sample")

# Page setup - A4
ALBUM_PAGES_SIZE (210.0 297.0)

# Margins (left, right, top, bottom)
ALBUM_PAGES_MARGINS (20.0 15.0 15.0 15.0)

# Double line border
ALBUM_PAGES_BORDER (0.5 0.1 0.0 1.0)

# Spacing between elements (horizontal, vertical)
ALBUM_PAGES_SPACING (8.0 8.0)

# Page title
ALBUM_PAGES_TITLE (TB 16 "Great Britain")

# Define a custom font
ALBUM_DEFINE_FONT (MY "DejaVu Serif")

# Set colours
COLOUR_ALBUM_TITLE (navy)
COLOUR_ALBUM_BORDER (black)

# ============================================================================
# Page 1 - The 1840 issue
# ============================================================================

PAGE_START

PAGE_TEXT_CENTRE (HB 12 "The First Postage Stamps")
PAGE_TEXT_CENTRE (HN 10 "Line-engraved, imperforate, 1840")

# Horizontal rule
PAGE_RULE_H (0.5 3 0)

PAGE_TEXT (HN 10 "Uniform Penny Postage began on 10 January 1840. Prepaid " \
                 "adhesive stamps followed in May: the Penny Black for " \
                 "letters up to half an ounce, and the Twopence Blue for " \
                 "letters up to one ounce.")

# Row of stamps
ROW_START_FS (HN 8 0.5 8.0)

STAMP_ADD (19.0 22.0 "1d black" "StampWorld 1" "" "")
STAMP_ADD (19.0 22.0 "2d blue" "StampWorld 2" "" "")

# ============================================================================
# Page 2 - The 1841 change of colour
# ============================================================================

PAGE_START

PAGE_TEXT_CENTRE (HB 12 "The Penny Red")
PAGE_TEXT_CENTRE (HN 10 "1841")

PAGE_RULE_H (0.3 3 0)

PAGE_TEXT (HN 10 "Red cancellations were hard to see on the Penny Black, " \
                 "and they could be cleaned off for re-use. In 1841 the " \
                 "penny stamp was printed in red-brown and cancelled in " \
                 "black ink instead.")

ROW_START_FS (HN 8 0.5 8.0)

STAMP_ADD (19.0 22.0 "1d red-brown" "StampWorld 3" "" "")
